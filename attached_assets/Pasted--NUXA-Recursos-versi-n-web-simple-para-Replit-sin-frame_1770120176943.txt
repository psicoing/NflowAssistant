const $ = (sel) => document.querySelector(sel);

const store = {
  get(key, fallback) {
    try {
      const raw = localStorage.getItem(key);
      return raw ? JSON.parse(raw) : fallback;
    } catch {
      return fallback;
    }
  },
  set(key, value) {
    localStorage.setItem(key, JSON.stringify(value));
  },
};

const todayISO = () => new Date().toISOString().slice(0, 10);
const daysBetween = (aISO, bISO) => {
  const a = new Date(aISO);
  const b = new Date(bISO);
  const ms = Math.abs(b - a);
  return Math.floor(ms / (1000 * 60 * 60 * 24));
};

function openModal(title, bodyHTML, footerButtons = []) {
  $("#modalTitle").textContent = title;
  $("#modalBody").innerHTML = bodyHTML;

  const footer = $("#modalFooter");
  footer.innerHTML = "";
  footerButtons.forEach((btn) => footer.appendChild(btn));

  $("#modal").hidden = false;
}

function closeModal() {
  $("#modal").hidden = true;
}

$("#modalClose").addEventListener("click", closeModal);
$("#modal").addEventListener("click", (e) => {
  if (e.target.id === "modal") closeModal();
});

// ----------------- Racha
function renderStreak() {
  const data = store.get("nuxa_streak", { days: 0, last: null });
  $("#streakDays").textContent = String(data.days);

  if (data.last) {
    $("#lastCheckInfo").textContent = "Último check-in: " + data.last;
  } else {
    $("#lastCheckInfo").textContent = "";
  }
}

$("#btnCheckIn").addEventListener("click", () => {
  const data = store.get("nuxa_streak", { days: 0, last: null });
  const t = todayISO();

  if (!data.last) {
    data.days = 1;
    data.last = t;
    store.set("nuxa_streak", data);
    renderStreak();
    return;
  }

  if (data.last === t) {
    // Ya marcado hoy
    return;
  }

  const gap = daysBetween(data.last, t);

  if (gap === 1) data.days += 1;
  else data.days = 1;

  data.last = t;
  store.set("nuxa_streak", data);
  renderStreak();
});

$("#btnResetStreak").addEventListener("click", () => {
  store.set("nuxa_streak", { days: 0, last: null });
  renderStreak();
});

// ----------------- Registro emocional + progreso
$("#btnSaveMood").addEventListener("click", () => {
  const mood = $("#moodSelect").value;
  const note = $("#moodNote").value.trim();
  const entry = { date: todayISO(), mood, note };

  const log = store.get("nuxa_mood_log", []);
  // Si ya hay entrada hoy, reemplaza (mejor que duplicar)
  const idx = log.findIndex((x) => x.date === entry.date);
  if (idx >= 0) log[idx] = entry;
  else log.push(entry);

  store.set("nuxa_mood_log", log);
  $("#moodNote").value = "";
  toast("Guardado ✅");
});

$("#btnViewProgress").addEventListener("click", () => {
  const log = store.get("nuxa_mood_log", []);
  if (!log.length) {
    openModal("📈 Mi evolución", "<p class='muted'>Aún no hay registros. Haz uno (30 segundos) y vuelve.</p>", [
      mkBtn("Cerrar", "ghost", closeModal),
    ]);
    return;
  }

  // Conteo simple por estados
  const count = {};
  log.forEach((e) => (count[e.mood] = (count[e.mood] || 0) + 1));

  const sorted = Object.entries(count).sort((a, b) => b[1] - a[1]);
  const fav = sorted.slice(0, 3).map(([k, v]) => `<li><b>${k}</b>: ${v}</li>`).join("");

  const last = log.slice(-8).reverse().map((e) => {
    const n = e.note ? ` — <span class="muted">${escapeHtml(e.note)}</span>` : "";
    return `<div style="padding:8px 10px; border:1px solid var(--line); border-radius:14px; margin:8px 0; background: rgba(0,0,0,.18)">
      <b>${e.date}</b> · ${emojiMood(e.mood)} ${e.mood}${n}
    </div>`;
  }).join("");

  openModal(
    "📈 Mi evolución",
    `
      <p class="muted">Resumen simple (sin nube, sin historias raras):</p>
      <div style="display:flex; gap:14px; flex-wrap:wrap">
        <div style="flex:1; min-width:220px; padding:10px 12px; border:1px solid var(--line); border-radius:14px; background: rgba(0,0,0,.18)">
          <b>Total registros:</b> ${log.length}<br/>
          <b>Top estados:</b>
          <ul style="margin:8px 0 0 18px">${fav}</ul>
        </div>
        <div style="flex:2; min-width:260px">
          <b>Últimos registros</b>
          ${last}
        </div>
      </div>
    `,
    [mkBtn("Cerrar", "ghost", closeModal)]
  );
});

// ----------------- ISO autochequeo
$("#btnIsoCalc").addEventListener("click", () => {
  const sliders = [...document.querySelectorAll("#isoForm input[type='range']")];
  // 0 = mal, 4 = bien (pero para riesgo conviene invertir)
  const values = sliders.map((s) => Number(s.value));
  const avg = values.reduce((a, b) => a + b, 0) / values.length; // 0..4
  const riskScore = 4 - avg; // 0..4 (más alto = más riesgo)

  let level = "Bajo";
  let rec = [
    "Mantener rutinas de descanso y límites razonables.",
    "Revisar carga de trabajo periódicamente.",
    "Refuerzo de reconocimiento y comunicación clara.",
  ];

  if (riskScore >= 1.6 && riskScore < 2.8) {
    level = "Medio";
    rec = [
      "Revisar carga, plazos y autonomía (ajustes rápidos).",
      "Formación breve en liderazgo y prevención de desgaste.",
      "Canal de conflictos seguro y trazable.",
    ];
  }

  if (riskScore >= 2.8) {
    level = "Alto";
    rec = [
      "Evaluación psicosocial formal y plan de acción con responsables.",
      "Medidas inmediatas: carga, horarios, rol y recursos.",
      "Intervención sobre liderazgo tóxico / cultura del miedo.",
      "Seguimiento con indicadores y revisiones periódicas.",
    ];
  }

  const badge = `<span class="isoBadge">${level}</span>`;
  const tip = `<p class="tiny muted">Basado en enfoque ISO 45003 (orientativo). Para decisiones, usar evaluación formal.</p>`;

  $("#isoResult").innerHTML = `
    ${badge}
    <p><b>Riesgo psicosocial estimado:</b> ${level}</p>
    <ul>${rec.map((x) => `<li>${x}</li>`).join("")}</ul>
    ${tip}
  `;
  $("#isoResult").hidden = false;
});

$("#btnIsoReset").addEventListener("click", () => {
  [...document.querySelectorAll("#isoForm input[type='range']")].forEach((s) => (s.value = 2));
  $("#isoResult").hidden = true;
});

// ----------------- Utilidades de ejercicios
function startBreath(seconds) {
  let remaining = seconds;
  openModal(
    "🫁 Respiración guiada",
    `
      <p><b>Inhala 4</b> · <b>Retén 2</b> · <b>Exhala 6</b></p>
      <div style="padding:12px; border:1px solid var(--line); border-radius:14px; background: rgba(0,0,0,.18); margin-top:10px">
        <div id="breathTimer" style="font-size:28px; font-weight:900">--:--</div>
        <div id="breathPhase" class="muted" style="margin-top:8px; font-weight:800"></div>
      </div>
      <p class="tiny muted" style="margin-top:10px">Si te mareas, baja el ritmo. Aquí no hay medallas.</p>
    `,
    [mkBtn("Terminar", "ghost", closeModal)]
  );

  const timerEl = $("#breathTimer");
  const phaseEl = $("#breathPhase");

  let phase = 0;
  const phases = [
    { name: "Inhala (4)", dur: 4 },
    { name: "Retén (2)", dur: 2 },
    { name: "Exhala (6)", dur: 6 },
  ];
  let phaseRemain = phases[0].dur;

  const tick = () => {
    const m = String(Math.floor(remaining / 60)).padStart(2, "0");
    const s = String(remaining % 60).padStart(2, "0");
    timerEl.textContent = `${m}:${s}`;
    phaseEl.textContent = phases[phase].name + ` · ${phaseRemain}s`;

    if (remaining <= 0) {
      phaseEl.textContent = "Hecho ✅";
      return;
    }

    remaining -= 1;
    phaseRemain -= 1;

    if (phaseRemain <= 0) {
      phase = (phase + 1) % phases.length;
      phaseRemain = phases[phase].dur;
    }

    if (!$("#modal").hidden) setTimeout(tick, 1000);
  };

  tick();
}

    `
      <p class="muted">Pantalla en calma. Solo respira y deja pasar.</p>

      </div>
    `,
    [mkBtn("Terminar", "ghost", closeModal)]
  );

  const tEl = $("#silenceTimer");

  const tick = () => {
    const m = String(Math.floor(remaining / 60)).padStart(2, "0");
    const s = String(remaining % 60).padStart(2, "0");
    tEl.textContent = `${m}:${s}`;

      return;

    remaining -= 1;
    if (!$("#modal").hidden) setTimeout(tick, 1000);
  };

  tick();

// ----------------- Pequeñas ayudas UI
function toast(text) {
  const d = document.createElement("div");
  d.textContent = text;
  d.style.position = "fixed";
  d.style.bottom = "18px";
  d.style.right = "18px";
  d.style.padding = "10px 12px";
  d.style.border = "1px solid var(--line)";
  d.style.borderRadius = "14px";
  d.style.background = "rgba(0,0,0,.55)";
  d.style.color = "var(--text)";
  d.style.fontWeight = "900";
  d.style.boxShadow = "var(--shadow)";
  document.body.appendChild(d);
  setTimeout(() => d.remove(), 1100);
}

function mkBtn(text, variant, onClick) {
  const b = document.createElement("button");
  b.className = "btn" + (variant ? " " + variant : "");
  b.textContent = text;
  b.addEventListener("click", onClick);
  return b;
}

function emojiMood(mood) {
  const map = {
    bien: "🙂",
    neutro: "😐",
    triste: "😔",
    ansioso: "😰",
    irritable: "😤",
    agotado: "🥱",
  };
  return map[mood] || "🙂";
}

function escapeHtml(str) {
  return str
    .replaceAll("&", "&amp;")
    .replaceAll("<", "&lt;")
    .replaceAll(">", "&gt;")
    .replaceAll('"', "&quot;")
    .replaceAll("'", "&#039;");
}
